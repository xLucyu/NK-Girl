export interface Command {
  commandData: SlashCommandOptionsOnlyBuilder | SlashCommandSubcommandsOnlyBuilder;
  execute(interaction: ChatInputCommandInteraction): Promise<void>;

  /** present ⇒ the command has autocompletable options */
  autoComplete?(interaction: AutocompleteInteraction): Promise<void>;

  /** present ⇒ the command owns message components */
  renderAndReply?(interaction: InteractionType, state: ComponentState): Promise<void>;
  handleModal?(state: ComponentState, key: string, input: string): boolean;
}

import {
  ButtonInteraction,
  MessageFlags,
  ModalSubmitInteraction,
  StringSelectMenuInteraction,
} from "discord.js";
import { componentState, scheduleComponentCleanup, TIMEOUT } from "@components";
import { commands, type Command } from "@commands";
import { applyPage, isPageValue } from "./pagination";
import { openModal, getModalInput } from "./modal";

type Trigger = ButtonInteraction | StringSelectMenuInteraction | ModalSubmitInteraction;

interface Context {
  messageId: string;
  state: NonNullable<ReturnType<typeof componentState.get>>;
  command: Command;
  field: string;
  value: string;
}

/** Guards + lookup, shared by components and modal submits. */
async function getContext(interaction: Trigger): Promise<Context | null> {

  const messageId = interaction.message?.id;
  if (!messageId) return null;

  const state = componentState.get(messageId);
  if (!state) return null;

  if (interaction.user.id !== state.userId) {
    await interaction.reply({
      content: "You're not the original user.",
      flags: MessageFlags.Ephemeral,
    });
    return null;
  }

  if (Date.now() > state.expiresAt) {
    componentState.delete(messageId);
    return null;
  }

  const [commandKey, field, value] = interaction.customId.split(":");
  const command = commands[commandKey];

  if (!command?.renderAndReply || !field) return null;

  return { messageId, state, command, field, value };
}

/** Refresh TTL, re-arm cleanup, re-render. */
async function commit(interaction: Trigger, context: Context): Promise<void> {

  const { messageId, state, command } = context;

  state.expiresAt = Date.now() + TIMEOUT;
  componentState.set(messageId, state);

  scheduleComponentCleanup({
    messageId,
    editReply: (options) => interaction.editReply(options),
    expiresAt: state.expiresAt,
    onExpire: () => componentState.delete(messageId),
  });

  try {
    await command.renderAndReply!(interaction, state);
  } catch (error) {
    console.error("[handleComponent] renderAndReply failed:", error);
    await interaction.followUp({
      content: "Something went wrong. Please try again.",
      flags: MessageFlags.Ephemeral,
    }).catch(() => undefined);
  }
}

export async function handleComponent(
  interaction: ButtonInteraction | StringSelectMenuInteraction,
): Promise<void> {

  const context = await getContext(interaction);
  if (!context) return;

  const { state, field } = context;
  const value = interaction.isStringSelectMenu() ? interaction.values[0] : context.value;

  // showModal IS the response — no deferUpdate, no commit
  if (field === "modal") {
    if (interaction.isButton()) await openModal(interaction, value);
    return;
  }

  await interaction.deferUpdate();

  if (field === "eventId") {
    state.event = value;
  } else if (isPageValue(value)) {
    applyPage(state, field, value);
  } else {
    state.options[field] = value;
  }

  await commit(interaction, context);
}

export async function handleModalSubmit(interaction: ModalSubmitInteraction): Promise<void> {

  const context = await getContext(interaction);
  if (!context) return;

  if (!context.command.handleModal) return;

  await interaction.deferUpdate();

  const changed = context.command.handleModal(
    context.state,
    context.value,                 // modal key, e.g. "lbSearch"
    getModalInput(interaction),
  );

  if (!changed) {
    await interaction.followUp({
      content: "No match found.",
      flags: MessageFlags.Ephemeral,
    }).catch(() => undefined);
    return;
  }

  await commit(interaction, context);
}

export const handleSelectMenu = handleComponent;
export const handleButton = handleComponent;
